# Pinned Frida toolchain for the capture hooks (tools/*.js via run_hook.py).
#
# INVARIANT: the host frida-tools version MUST EQUAL the frida-gadget version
# baked into the repackaged Aqara app. Frida broke wire compatibility between
# 16.x and 17.x, so a drifted host (e.g. an unpinned `pipx install frida-tools`
# that pulled 17.x against a 16.x gadget) fails the handshake with
# "connection closed" / "Failed to spawn". See tools/frida-setup.md.
#
# Install (host):  pip install -r tools/requirements-frida.txt
#              or:  pipx runpip frida-tools install --force-reinstall "frida==16.7.19"
#                   (frida-tools itself has its own, lower version numbering on
#                   PyPI — e.g. 14.10.4 — it's the CORE `frida` package's version
#                   that must match the gadget; `frida --version` on the CLI
#                   reports the core version, not the frida-tools package version)
frida==16.7.19
